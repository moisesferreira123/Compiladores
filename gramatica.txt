+ EXPRESSÕES ARITMÉTICAS

EXP ::= TERM EXP'
EXP' ::= + TERM EXP'
EXP' ::= - TERM EXP'
EXP' ::= ''
TERM ::= POW TERM'
TERM' ::= * POW TERM'
TERM' ::= / POW TERM'
TERM' ::= ''
POW ::= FACT POW'
POW' ::= ^ POW
POW' ::= ''
FACT ::= ( EXP )
FACT ::= id
