+ EXPRESSÕES ARITMÉTICAS

EXP ::= TERM EXP'
EXP' ::= + TERM EXP'
EXP' ::= - TERM EXP'
EXP' ::= ''
TERM ::= POW TERM'
TERM' ::= * POW TERM'
TERM' ::= / POW TERM'
TERM' ::= ''
POW ::= FACT POW'
POW' ::= ^ POW
POW' ::= ''
FACT ::= ( EXP )
FACT ::= id

+ EXPRESSÕES RELACIONAIS

EXP ::= EXP_ARITH EXP_REL
EXP_REL ::= REL_OP EXP_ARITH 
EXP_REL ::= ''
EXP_ARITH ::= TERM EXP_ARITH'
EXP_ARITH' ::= + TERM EXP_ARITH'
EXP_ARITH' ::= - TERM EXP_ARITH'
EXP_ARITH' ::= ''
TERM ::= POW TERM'
TERM' ::= * POW TERM'
TERM' ::= / POW TERM'
TERM' ::= ''
POW ::= FACT POW'
POW' ::= ^ POW
POW' ::= ''
FACT ::= ( EXP )
FACT ::= id
REL_OP ::= <
REL_OP ::= <=
REL_OP ::= >
REL_OP ::= >=
REL_OP ::= =
REL_OP ::= <>
