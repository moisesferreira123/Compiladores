program swapexemple
begin
  procedure swap (a: ref(int), b: ref(int))
    begin
      var tmp: int
    in
      tmp := deref(a);
      deref(a) := deref(b);
      deref(b) := tmp
    end;
  procedure main()
    begin
      var x: int;
      var y: int;
      var xr: ref(int);
      var yr: ref(int)
    in
      x := 42; y := 84;
      if x = y then
        tmp := -x + y
      else 
        x := x^y;
        x := x*y
      fi;
      xr := ref(x); yr := ref(y);
      swap(xr,yr)
    end
end
